#include <iostream>
#include <bitset>

// Setting Bit - (val | 00000100)
// Resetting Bit - (val & 11111011)
// Checking Bit - (val & 00000100)
// Toggling Bit - (val ^ 00000100)

int main() {
    int val = 0b01001100;

    // _____________Set Bit______________________________________________________________________________________________
    std::cout << std::bitset<8>(val | (1 << 0)) << std::endl; // setting bit at position 0 only.
    std::cout << std::bitset<8>(val | (1 << 5)) << std::endl; // setting bit at position 5 only.
    std::cout << std::bitset<8>(val | ((1 << 0) | (1 << 5))) << std::endl; // setting bits at position 0 and 5 both.

    // _____________Reset Bit____________________________________________________________________________________________
    std::cout << std::bitset<8>(val & ~(1 << 2)) << std::endl; // resetting bit at position 2 only.
    std::cout << std::bitset<8>(val & ~(1 << 6)) << std::endl; // resetting bit at position 6 only.
    std::cout << std::bitset<8>(val & ~((1 << 2) | (1 << 6))) << std::endl; // resetting bits at position 2 and 6 both.

    // _____________Check Bit____________________________________________________________________________________________
    std::cout << std::boolalpha;
    std::cout << (((val & (1 << 0)) >> 0) == 1) << std::endl; // checking bit at position 0 only.
    std::cout << (((val & (1 << 3)) >> 3) == 1) << std::endl; // checking bit at position 3 only.
    std::cout << std::noboolalpha;

    // _____________Toggle Bit___________________________________________________________________________________________
    std::cout << std::bitset<8>(val ^ (1 << 0)) << std::endl; // toggle bit at position 0 only.
    std::cout << std::bitset<8>(val ^ (1 << 3)) << std::endl; // toggle bit at position 3 only.
    std::cout << std::bitset<8>(val ^ ((1 << 0) | (1 << 3))) << std::endl; // toggle bits at position 0 and 3 both.

    return 0;
}
